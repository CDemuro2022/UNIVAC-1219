ORG &O10000
SADD &O10000
JP MAIN
;
; A program which computes the digits of Pi and prints them in the shape of a PI.
;
; The RJP PRINT_SHAPE lines can be replaced with RJP PRINT_CHAR to get a regular
; rectangular printout.
;
; Enable stop switch 1 to make the program stop when finished.
;
;
;
;
; EXFAL
; Performs the external function contained in AL.
;
EVEN
>EXFWD DATA 0
>EXFAL DATA 0
STRAL EXFWD
EXF 0
DATA EXFWD
DATA EXFWD
SKPFIN 0
JP LOK-1
IJP EXFAL
;
;
;
; OUTAL
; Prints the contents of AL to output as a character.
;
>OUTWD DATA 0
>OUTAL DATA 0
STRAL OUTWD
OUT 0
DATA OUTWD
DATA OUTWD
SKPOIN 0
JP LOK-1
IJP OUTAL
;
;
;
; DOT
; Prints a period character "." to output.
;
>DOT DATA 0
ENTALK &56
RJP PRINT_SHAPE
;RJP PRINT_CHAR
IJP DOT
;
;
;
; CRLF
; Prints a carriage-return, line-feed sequence output.
;
>CRLF DATA 0
ENTALK &15
RJP OUTAL
ENTALK &12
RJP OUTAL
IJP CRLF
;
;
;
; PRINT_CHAR
; Prints the character in AL to output. If the carriage reaches column 72
; (as counted by LINE_CNT) also print a "\r\n"
;
>LINE_CNT DATA &107
>PRINT_CHAR DATA 0
RJP OUTAL
ISK LINE_CNT
JP PRINT_CHAR_DONE
ENTALK &107
STRAL LINE_CNT
RJP CRLF
>PRINT_CHAR_DONE IJP PRINT_CHAR
;
;
;
; DECODE_RUN
; Decodes a word from the run-length shape encoding
;
>SHAPE_RUN_IDX DATA 0
>BYTE_MASK DATA 255
>BIT_MASK DATA 1
>RUN DATA 0
>DECODE_RUN DATA 0
ENTB SHAPE_RUN_IDX
ENTALB SHAPE_RUN_DATA
STRAL RUN
SLCL BYTE_MASK
STRAL RUN_FILLED
ENTAL RUN
RSHAL 9
SLCL BYTE_MASK
STRAL RUN_SPACES
ENTAL RUN
RSHAL 17
SLCL BIT_MASK
STRAL RUN_NEWLINE
IJP DECODE_RUN
;
;
;
; PRINT_SHAPE
; Prints characters as defined in the shape 
;
>RUN_FILLED DATA 0
>RUN_SPACES DATA 0
>RUN_NEWLINE DATA 0
>CHAR DATA 0
>PRINT_SHAPE DATA 0
STRAL CHAR
; Check if there's a run of filled characters to do, if not, decode one
ISK RUN_FILLED
JP DONE_DECODING
RJP DECODE_RUN
ENTAL RUN_SPACES
JPALZ DONE_DECODING
ISK RUN_SPACES
JP LOK+1
ENTALK &040
>SPACE_LOOP RJP OUTAL
ISK RUN_SPACES
JP SPACE_LOOP
; After we decode and print the space run, print the character
>DONE_DECODING ENTAL CHAR
RJP OUTAL
; If this was the last filled character in the run, bump the index and
; maybe print a newline
ENTAL RUN_FILLED
SUBAL BIT_MASK
JPALNZ EXIT_PRINT_SHAPE
CL RUN_FILLED
ENTAL SHAPE_RUN_IDX
ADDALK 1
STRAL SHAPE_RUN_IDX
; Check if we're on the last run, and exit
CMAL SHAPE_RUN_CNT
JPALNZ LOK+2
STOP 1
ENTAL RUN_NEWLINE
JPALZ EXIT_PRINT_SHAPE
RJP CRLF
>EXIT_PRINT_SHAPE IJP PRINT_SHAPE
;
;
;
; EXTRACT_DIGIT
; Removes the integer part of the PI buffer and prints the digit to output.
; Multiplies the ACCUM and NUMER buffers by 10. If this was called for the 
; first time, also prints a dot after the digit
;
>PRINTED_DOT DATA 0
>EXTRACT_DIGIT DATA 0
ENTAL PI
ADDALK &060
RJP PRINT_SHAPE
;RJP PRINT_CHAR
CL PI
RJP MUL_TEN_ACCUM
RJP MUL_TEN_NUMER
; Only print a dot if PRINTED_DOT is false
ENTAL PRINTED_DOT
JPALNZ AFTER_DOT
RJP DOT
ENTALK 1
STRAL PRINTED_DOT
>AFTER_DOT IJP EXTRACT_DIGIT
;
;
;
; MUL_TEN_NUMER
; Multiplies the contents of the NUMER buffer by 10.
;
>ZERO DATA 0
>NEG_ZERO DATA &O777777
>IDX DATA 0
EVEN
>ADDA_CORR DATA 3
DATA 0
EVEN
>ADDA_CORR_2 DATA 2
DATA 0
EVEN
>ACCUML DATA 0
>ACCUMU DATA 0
>CARRYL DATA 0
>CARRYU DATA 0
>MUL_TEN_NUMER DATA 0
ENTAL DIGIT_CNT
CL CARRYL
CL CARRYU
STRAL IDX
ENTAL ZERO
ENTAU ZERO
>TEN_NUMER_LOOP ENTB IDX
ENTAU ZERO
ENTALB NUMER
STRAU ACCUMU
STRAL ACCUML
LSHA 2
ADDA ACCUML
LSHA 1
ADDA CARRYL
SUBA ADDA_CORR
STRALB NUMER
STRAU CARRYL
ISK IDX
JP TEN_NUMER_LOOP
IJP MUL_TEN_NUMER
;
;
;
; MUL_TEN_ACCUM
; Multiplies the contents of the PI buffer by 10.
;
>MUL_TEN_ACCUM DATA 0
CL CARRYL
CL CARRYU
ENTAL DIGIT_CNT
STRAL IDX
ENTAL ZERO
ENTAU ZERO
>TEN_ACCUM_LOOP ENTB IDX
ENTAU ZERO
ENTALB PI
STRAU ACCUMU
STRAL ACCUML
LSHA 2
ADDA ACCUML
LSHA 1
ADDA CARRYL
SUBA ADDA_CORR
STRALB PI
STRAU CARRYL
ISK IDX
JP TEN_ACCUM_LOOP
IJP MUL_TEN_ACCUM
;
;
;
; NUMER_SHIFT
; Shifts the contents of the NUMER buffer 10 bits to the right. This is 
; equivalent to dividing by 1024.
;
>COUNT DATA 0
>NUMER_SHIFT DATA 0
ENTAL DIGIT_CNT
STRAL COUNT
>NUMER_SHIFT_LOOP ENTB COUNT
ENTALB NUMER
ENTAUB NUMER_UPPER
RSHA 10
STRALB NUMER
ISK COUNT
JP NUMER_SHIFT_LOOP
IJP NUMER_SHIFT
;
;
;
; INIT_NUMER
; Initializes the value in the NUMER and PI buffers. Both are zeroed out except
; for the first value in NUMER which is set to 4
;
>INIT_NUMER DATA 0
ENTAL DIGIT_CNT
STRAL COUNT
>INIT_NUMER_LOOP ENTB COUNT
CLB NUMER
CLB PI
ISK COUNT
JP INIT_NUMER_LOOP
ENTALK 4
STRAL NUMER
IJP INIT_NUMER
;
;
;
; ACCUMULATE
; Adds the values in the QUOTIENT buffer to the PI buffer
;
EVEN
>PIL DATA 0
>PIU DATA 0
>QL DATA 0
>QU DATA 0
>CARRYL DATA 0
>CARRYU DATA 0
>ACCUMULATE DATA 0
ENTAL DIGIT_CNT
STRAL COUNT
CL CARRYL
CL CARRYU
CL PIU
CL PIL
>ACCUMULATE_LOOP ENTB COUNT
ENTALB PI
STRAL PIL
ENTALB QUOTIENT
STRAL QL
ENTAU ZERO
ENTAL ZERO
ADDA CARRYL
ADDA PIL
ADDA QL
SUBA ADDA_CORR
STRALB PI
STRAU CARRYL
ISK COUNT
JP ACCUMULATE_LOOP
IJP ACCUMULATE
;
;
;
; ACCUMULATE_SUB
; Subtracts the value in the QUOTIENT buffer from the PI buffer
;
>ACCUMULATE_SUB DATA 0
ENTAL DIGIT_CNT
STRAL COUNT
CL CARRYL
CL CARRYU
CL PIU
CL PIL
; Even though the hardware is 1's complement, we try to implement this as
; a 2's complement subtraction. We need to start with a 1 in the carry bit
ENTALK 1
STRAL CARRYL
>ACC_SUB_LOOP ENTB COUNT
ENTALB PI
STRAL PIL
ENTAU ZERO
ENTALB QUOTIENT
; CPAL does nothing for zero, we need to special case to fill this one with
; ones
JPALNZ DO_CPAL
ENTAL NEG_ZERO
JP LOK+2
>DO_CPAL CPAL 0
ADDA PIL
ADDA CARRYL
SUBA ADDA_CORR_2
STRALB PI
STRAU CARRYL
ISK COUNT
JP ACC_SUB_LOOP
IJP ACCUMULATE_SUB
;
;
;
; DIVIDE
; Divides the contents of the NUMER buffer by the divisor in DENOM
; Writes the output to the QUOTIENT buffer for DIGIT_CNT words
;
>OUTER_COUNT DATA 0
>DENOM DATA 1
>DIVIDE DATA 0
ENTAL DIGIT_CNT
STRAL OUTER_COUNT
ENTBK 0
ENTALK 0
RSHA 18
>DIVIDE_OUTER_LOOP ENTALB NUMER
DIVA DENOM
STRALB QUOTIENT
ENTBKB 1
ISK OUTER_COUNT
JP DIVIDE_OUTER_LOOP
IJP DIVIDE
;
;
;
; QUOTIENT_SHIFT
; Shifts the contents of the QUOTIENT buffer to the right. The number 
; of bits is determined by the contents of the AL register.
;
>ZERO_SHIFT_INSTRUCTION RSHA 0
>QUOTIENT_SHIFT DATA 0
SLSET ZERO_SHIFT_INSTRUCTION
STRAL SHIFT_INSTRUCTION
ENTAL DIGIT_CNT
STRAL COUNT
>QUOTIENT_SHIFT_LOOP ENTB COUNT
ENTALB QUOTIENT
ENTAUB QUOTIENT_UPPER
>SHIFT_INSTRUCTION RSHA 0
STRALB QUOTIENT
ISK COUNT
JP QUOTIENT_SHIFT_LOOP
IJP QUOTIENT_SHIFT
;
;
;
; MAIN 
; The main loop of the program
;
>STOP_ITER DATA 300
>DIGIT_CNT DATA &236
>K DATA 0
>K4 DATA 0
>K10 DATA 0
>MAIN SIL 0
ENTSR 0
RJP INIT_NUMER
ENTALK &O13
RJP EXFAL
; Compute K, K * 4, and K * 10 for the denominators
>LOOP ENTAL K
LSHAL 2
STRAL K4
ADDAL K
LSHAL 1
STRAL K10
;
;
; The even iterations of the summation (where -1^k == 1)
ENTAL K10
ADDALK 1
STRAL DENOM
RJP DIVIDE
RJP ACCUMULATE
;
ENTAL K4
ADDALK 1
LSHAL 3
STRAL DENOM
RJP DIVIDE
RJP ACCUMULATE_SUB
;
ENTAL K4
ADDALK 3
STRAL DENOM
RJP DIVIDE
ENTALK 8
RJP QUOTIENT_SHIFT
RJP ACCUMULATE_SUB
;
ENTAL K10
ADDALK 3
LSHAL 2
STRAL DENOM
RJP DIVIDE
RJP ACCUMULATE_SUB
;
ENTAL K10
ADDALK 5
STRAL DENOM
RJP DIVIDE
ENTALK 6
RJP QUOTIENT_SHIFT
RJP ACCUMULATE_SUB
;
ENTAL K10
ADDALK 7
STRAL DENOM
RJP DIVIDE
ENTALK 6
RJP QUOTIENT_SHIFT
RJP ACCUMULATE_SUB
;
ENTAL K10
ADDALK 9
STRAL DENOM
RJP DIVIDE
ENTALK 8
RJP QUOTIENT_SHIFT
RJP ACCUMULATE
;
ENTAL K
CMAL STOP_ITER
JPALNZ LOK+2
STOP 1
;
; Extract and print digits. (If this is the first iteration, print a dot)
RJP EXTRACT_DIGIT
RJP EXTRACT_DIGIT
RJP EXTRACT_DIGIT
RJP NUMER_SHIFT
;
;
ENTAL K
ADDALK 1
STRAL K
;
; Compute K, K * 4, and K * 10 for the denominators
ENTAL K
LSHAL 2
STRAL K4
ADDAL K
LSHAL 1
STRAL K10
;
;
; The odd iterations of the summation (where -1^k == -1)
ENTAL K4
ADDALK 1
LSHAL 3
STRAL DENOM
RJP DIVIDE
RJP ACCUMULATE
;
ENTAL K4
ADDALK 3
STRAL DENOM
RJP DIVIDE
ENTALK 8
RJP QUOTIENT_SHIFT
RJP ACCUMULATE
;
ENTAL K10
ADDALK 1
STRAL DENOM
RJP DIVIDE
RJP ACCUMULATE_SUB
;
ENTAL K10
ADDALK 3
LSHAL 2
STRAL DENOM
RJP DIVIDE
RJP ACCUMULATE
;
ENTAL K10
ADDALK 5
STRAL DENOM
RJP DIVIDE
ENTALK 6
RJP QUOTIENT_SHIFT
RJP ACCUMULATE
;
ENTAL K10
ADDALK 7
STRAL DENOM
RJP DIVIDE
ENTALK 6
RJP QUOTIENT_SHIFT
RJP ACCUMULATE
;
ENTAL K10
ADDALK 9
STRAL DENOM
RJP DIVIDE
ENTALK 8
RJP QUOTIENT_SHIFT
RJP ACCUMULATE_SUB
;
ENTAL K
CMAL STOP_ITER
JPALNZ LOK+2
STOP 1
;
RJP EXTRACT_DIGIT
RJP EXTRACT_DIGIT
RJP EXTRACT_DIGIT
RJP NUMER_SHIFT
;
;
ENTAL K
ADDALK 1
STRAL K
JP LOOP
;
